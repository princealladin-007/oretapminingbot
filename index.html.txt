<script>
(function() {
  'use strict';

  const CONFIG = {
    ORE_PER_100_TAPS: 50,
    EXTRA_TAPS_PER_AD: 10,
    ORE_PER_EXTRA_TAP: 0.5,
    COOLDOWN_HOURS: 24,
    BASE_TAPS: 100,
    STORAGE_KEY: 'oreMineDataV9'
  };

  // ============================================================
  // 🔥 MONETAG CONFIG — Zone ID: 11873387
  // ============================================================
  const MONETAG_FN_NAME = 'show_11873387';

  let state = {
    ore: 0,
    totalTaps: 0,
    baseTaps: 0,
    extraTapsUsed: 0,
    dailyEarned: 0,
    extraTaps: 0,
    cooldownUntil: 0,
    locked: false
  };

  function loadState() {
    try {
      const saved = localStorage.getItem(CONFIG.STORAGE_KEY);
      if (saved) {
        const parsed = JSON.parse(saved);
        state = Object.assign(state, parsed);
      }
    } catch(e) { console.warn('Load error:', e); }
  }

  function saveState() {
    try {
      localStorage.setItem(CONFIG.STORAGE_KEY, JSON.stringify(state));
    } catch(e) { console.warn('Save error:', e); }
  }

  const oreCountEl = document.getElementById('oreCount');
  const tapCountEl = document.getElementById('tapCount');
  const dailyEarnedEl = document.getElementById('dailyEarned');
  const extraTapsEl = document.getElementById('extraTaps');
  const tapBtn = document.getElementById('tapBtn');
  const tapArea = document.getElementById('tapArea');
  const watchAdBtn = document.getElementById('watchAdBtn');
  const listBtn = document.getElementById('listBtn');
  const modalOverlay = document.getElementById('modalOverlay');
  const modalTitle = document.getElementById('modalTitle');
  const modalMessage = document.getElementById('modalMessage');
  const modalReward = document.getElementById('modalReward');
  const adProgress = document.getElementById('adProgress');
  const adProgressBar = document.getElementById('adProgressBar');
  const modalCloseBtn = document.getElementById('modalCloseBtn');
  const toastEl = document.getElementById('toast');
  const cooldownBanner = document.getElementById('cooldownBanner');
  const cooldownTimer = document.getElementById('cooldownTimer');
  const progressBar = document.getElementById('progressBar');
  const progressText = document.getElementById('progressText');

  function formatOre(value) {
    if (Number.isInteger(value)) return value.toString();
    return value.toFixed(1);
  }

  function getTotalTaps() {
    return state.baseTaps + state.extraTapsUsed;
  }

  function getTotalAllowed() {
    return CONFIG.BASE_TAPS + state.extraTaps + state.extraTapsUsed;
  }

  function updateUI() {
    const totalTaps = getTotalTaps();
    const totalAllowed = getTotalAllowed();

    oreCountEl.textContent = formatOre(state.ore);
    tapCountEl.textContent = totalTaps;
    dailyEarnedEl.textContent = formatOre(state.dailyEarned);
    extraTapsEl.textContent = state.extraTaps;

    const percent = Math.min((totalTaps / totalAllowed) * 100, 100);
    progressBar.style.width = percent + '%';
    progressText.textContent = `${totalTaps} / ${totalAllowed} taps`;
  }

  let toastTimer;
  function showToast(msg, type) {
    toastEl.textContent = msg;
    let cls = 'toast show';
    if (type === 'error') cls += ' error';
    else if (type === 'warning') cls += ' warning';
    toastEl.className = cls;
    clearTimeout(toastTimer);
    toastTimer = setTimeout(() => {
      toastEl.className = 'toast';
    }, 2500);
  }

  function showFloatText(x, y, text) {
    const el = document.createElement('div');
    el.className = 'float-text';
    el.textContent = text;
    el.style.left = x + 'px';
    el.style.top = y + 'px';
    el.style.position = 'fixed';
    tapArea.appendChild(el);
    setTimeout(() => el.remove(), 800);
  }

  function formatTime(ms) {
    if (ms <= 0) return '00:00:00';
    const totalSec = Math.floor(ms / 1000);
    const h = Math.floor(totalSec / 3600);
    const m = Math.floor((totalSec % 3600) / 60);
    const s = totalSec % 60;
    return String(h).padStart(2, '0') + ':' +
           String(m).padStart(2, '0') + ':' +
           String(s).padStart(2, '0');
  }

  function updateLockUI() {
    const now = Date.now();
    const totalAllowed = getTotalAllowed();
    const totalTaps = getTotalTaps();

    if (state.locked && state.cooldownUntil <= now) {
      state.locked = false;
      state.cooldownUntil = 0;
      state.baseTaps = 0;
      state.extraTapsUsed = 0;
      state.dailyEarned = 0;
      state.extraTaps = 0;
      saveState();
      showToast('🌅 Cooldown ended! Mine again.');
    }

    if (state.locked && state.cooldownUntil > now) {
      cooldownBanner.classList.add('active');
      cooldownTimer.textContent = formatTime(state.cooldownUntil - now);
    } else {
      cooldownBanner.classList.remove('active');
    }

    if (totalTaps < totalAllowed) {
      tapBtn.disabled = false;
      tapBtn.classList.remove('locked');
    } else {
      tapBtn.disabled = true;
      tapBtn.classList.add('locked');
    }

    watchAdBtn.disabled = false;
    if (state.locked && state.cooldownUntil > now) {
      watchAdBtn.classList.add('pulse-active');
    } else {
      watchAdBtn.classList.remove('pulse-active');
    }

    updateUI();
  }

  setInterval(updateLockUI, 1000);

  function handleTap(e) {
    e.preventDefault();

    const totalAllowed = getTotalAllowed();
    const totalTaps = getTotalTaps();

    if (totalTaps >= totalAllowed) {
      showToast('⚠️ Tap limit reached! Watch ad for +10 taps.', 'warning');
      return;
    }

    state.totalTaps++;

    let isExtraTap = false;

    if (state.baseTaps < CONFIG.BASE_TAPS) {
      state.baseTaps++;

      if (state.baseTaps === CONFIG.BASE_TAPS) {
        state.ore += CONFIG.ORE_PER_100_TAPS;
        state.dailyEarned += CONFIG.ORE_PER_100_TAPS;
        state.locked = true;
        state.cooldownUntil = Date.now() + (CONFIG.COOLDOWN_HOURS * 60 * 60 * 1000);
        saveState();

        showToast(`🎉 +${CONFIG.ORE_PER_100_TAPS} Ore! Locked for 24h.`, 'warning');

        modalTitle.textContent = '🔒 Mining Locked!';
        modalMessage.textContent = `You completed ${CONFIG.BASE_TAPS} taps and earned ${CONFIG.ORE_PER_100_TAPS} ore!\n\n👉 Watch Ad below to add +10 extra taps within this 24h cycle.`;
        modalReward.textContent = `+${CONFIG.ORE_PER_100_TAPS} 💎`;
        adProgress.style.display = 'none';
        modalCloseBtn.style.display = 'inline-block';
        modalOverlay.classList.add('active');

        updateLockUI();
        return;
      }
    } else {
      isExtraTap = true;
      state.extraTapsUsed++;
      if (state.extraTaps > 0) {
        state.extraTaps--;
      }
      state.ore += CONFIG.ORE_PER_EXTRA_TAP;
      state.dailyEarned += CONFIG.ORE_PER_EXTRA_TAP;
    }

    const rect = tapBtn.getBoundingClientRect();
    const x = rect.left + rect.width / 2 + (Math.random() - 0.5) * 60;
    const y = rect.top + rect.height / 2 + (Math.random() - 0.5) * 60;
    const remaining = totalAllowed - (totalTaps + 1);
    const text = isExtraTap
      ? `+${CONFIG.ORE_PER_EXTRA_TAP} 💎 (${remaining} left)`
      : `+⛏️ (${remaining} left)`;
    showFloatText(x, y, text);

    saveState();
    updateLockUI();
  }

  tapBtn.addEventListener('click', handleTap);
  tapBtn.addEventListener('touchstart', function(e) {
    e.preventDefault();
    handleTap(e);
  }, { passive: false });

  // ============================================================
  // 📺 WATCH AD — Monetag (Zone ID: 11873387)
  // ============================================================
  let adPlaying = false;

  watchAdBtn.addEventListener('click', function() {
    if (adPlaying) return;

    adPlaying = true;
    watchAdBtn.disabled = true;
    watchAdBtn.textContent = '⏳ Loading Ad...';

    // Monetag ফাংশন চেক
    if (typeof window[MONETAG_FN_NAME] !== 'function') {
      adPlaying = false;
      watchAdBtn.disabled = false;
      watchAdBtn.textContent = '📺 Watch Ad (+10 Extra Taps)';
      showToast('❌ Ad SDK not loaded. Check connection.', 'error');
      console.error('Monetag function not found:', MONETAG_FN_NAME);
      return;
    }

    // Loading মডাল দেখাও
    modalTitle.textContent = '📺 Advertisement';
    modalMessage.textContent = 'Please wait while the ad is loading...';
    modalReward.textContent = '';
    adProgress.style.display = 'block';
    adProgressBar.style.width = '0%';
    modalCloseBtn.style.display = 'none';
    modalOverlay.classList.add('active');

    // Monetag অ্যাড কল
    window[MONETAG_FN_NAME]()
      .then(function(result) {
        // ✅ অ্যাড সফল
        console.log('Monetag ad completed:', result);

        adProgress.style.display = 'none';
        modalCloseBtn.style.display = 'inline-block';
        adPlaying = false;

        // Reward
        state.extraTaps += CONFIG.EXTRA_TAPS_PER_AD;
        saveState();

        modalTitle.textContent = '🎉 Ad Completed!';
        modalMessage.textContent = `You earned ${CONFIG.EXTRA_TAPS_PER_AD} extra taps!\n\nEach extra tap gives ${CONFIG.ORE_PER_EXTRA_TAP} ore. You can tap ${CONFIG.EXTRA_TAPS_PER_AD} more times.`;
        modalReward.textContent = `+${CONFIG.EXTRA_TAPS_PER_AD} ⛏️`;

        watchAdBtn.disabled = false;
        watchAdBtn.textContent = '📺 Watch Ad (+10 Extra Taps)';

        showToast(`✅ +${CONFIG.EXTRA_TAPS_PER_AD} extra taps added!`);
        updateLockUI();
      })
      .catch(function(error) {
        // ❌ অ্যাড ফেইল / ইউজার বন্ধ করেছে
        console.warn('Monetag ad error:', error);

        adProgress.style.display = 'none';
        modalCloseBtn.style.display = 'inline-block';
        adPlaying = false;

        modalTitle.textContent = '❌ Ad Failed';
        modalMessage.textContent = 'The ad could not be shown. Please try again later.';
        modalReward.textContent = '';

        watchAdBtn.disabled = false;
        watchAdBtn.textContent = '📺 Watch Ad (+10 Extra Taps)';

        showToast('❌ Ad failed. Try again.', 'error');
      });
  });

  listBtn.addEventListener('click', function() {
    if (state.ore <= 0) {
      showToast('❌ No ore to list!', 'error');
      return;
    }

    modalTitle.textContent = '📋 List Ore for Sale';
    modalMessage.textContent = `You are listing ${formatOre(state.ore)} ore on the marketplace. Once listed, it will be available for trading.`;
    modalReward.textContent = `${formatOre(state.ore)} 💎`;
    adProgress.style.display = 'none';
    modalCloseBtn.style.display = 'inline-block';
    modalOverlay.classList.add('active');
  });

  modalCloseBtn.addEventListener('click', function() {
    modalOverlay.classList.remove('active');
    updateLockUI();
  });

  modalOverlay.addEventListener('click', function(e) {
    if (e.target === modalOverlay) {
      modalOverlay.classList.remove('active');
      updateLockUI();
    }
  });

  loadState();
  updateLockUI();

  if (window.Telegram && window.Telegram.WebApp) {
    const tg = window.Telegram.WebApp;
    tg.ready();
    tg.expand();
    tg.setHeaderColor('#1a1a2e');
    tg.setBackgroundColor('#1a1a2e');
  }

})();
